```base
formulas:
  common_tags: 'file.tags.filter(this.file.tags.contains(value))'
  other_tags: 'file.tags.filter(!this.file.tags.contains(value))'

properties:
  file.name:
    displayName: 노트
  formula.common_tags:
    displayName: 공통태그
  formula.other_tags:
    displayName: 그외태그

views:
  - type: table
    name: 공통 태그 노트
    filters:
      and:
        - 'file.path != this.file.path'
        - '!formula.common_tags.isEmpty()'
    order:
      - file.name
      - formula.common_tags
      - formula.other_tags
```
